# Jazzy wrapper and atomic polynomial/result interfaces live in the same package.
cmake_minimum_required(VERSION 3.16)
project(path_planning LANGUAGES C CXX)
if(POLICY CMP0144)
  cmake_policy(SET CMP0144 NEW)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(path_searching REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/PolynomialSegment.msg" "msg/PlanResult.msg"
  DEPENDENCIES std_msgs builtin_interfaces trajectory_msgs)
add_executable(path_planning_node src/path_planning.cpp)
target_compile_features(path_planning_node PRIVATE cxx_std_17)
target_compile_options(path_planning_node PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(path_planning_node rclcpp path_searching pcl_conversions
  nav_msgs sensor_msgs geometry_msgs trajectory_msgs)
target_link_libraries(path_planning_node path_searching::kinodynamic_astar)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(path_planning_node "${typesupport}")
install(TARGETS path_planning_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(ros_io test/test_ros_io.py TIMEOUT 90)
endif()
ament_package()
