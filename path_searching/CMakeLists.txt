# ROS-independent numerical library, packaged with ament for Jazzy.
cmake_minimum_required(VERSION 3.16)
project(path_searching LANGUAGES C CXX)
if(POLICY CMP0144)
  cmake_policy(SET CMP0144 NEW)
endif()
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters search kdtree)
add_library(kinodynamic_astar SHARED src/kinodynamic_astar.cpp)
target_compile_features(kinodynamic_astar PUBLIC cxx_std_17)
target_compile_options(kinodynamic_astar PRIVATE -Wall -Wextra -Wpedantic -Werror=return-type)
target_include_directories(kinodynamic_astar PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>
  ${PCL_INCLUDE_DIRS})
target_link_libraries(kinodynamic_astar PUBLIC Eigen3::Eigen ${PCL_LIBRARIES})
option(ENABLE_SANITIZERS "Instrument this package and tests with ASan/UBSan" OFF)
if(ENABLE_SANITIZERS)
  # Ubuntu's prebuilt PCL uses glibc's 16-byte-aligned malloc. Eigen switches allocator
  # under __SANITIZE_ADDRESS__; keep the ABI identical across the shared-library boundary.
  target_compile_definitions(kinodynamic_astar PUBLIC EIGEN_MALLOC_ALREADY_ALIGNED=1)
  target_compile_options(kinodynamic_astar PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(kinodynamic_astar PUBLIC -fsanitize=address,undefined)
endif()
install(TARGETS kinodynamic_astar EXPORT export_kinodynamic_astar LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_kinodynamic_astar HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3 PCL)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_search test/test_search.cpp)
  target_link_libraries(test_search kinodynamic_astar)
endif()
ament_package()
